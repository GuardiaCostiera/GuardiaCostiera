/******************************************************************************* * SYSTEM: NAV-SENTINEL v3.81 (Build 1998-11-04) * SUBSYSTEM: COAST GUARD MARITIME SURVEILLANCE & SAR ENGINE * MODULE: RADAR_TRACKER.C * AUTHOR: SYS_ADMIN // DEPT OF MARITIME SECURITY * * DESCRIPTION: Core tracking module for target identification, SAR sector * sweep routing, and automated MAYDAY telemetry parsing. * Written for ANSI C / POSIX compliance on UNIX workstations. ******************************************************************************/ #include #include #include #include /* --- CONSTANTS & SYSTEM PARAMETERS --- */ #define MAX_TARGETS 256 #define RADAR_RANGE_NM 48.00f /* Radar max range in Nautical Miles */ #define RADAR_SWEEP_FREQ_HZ 12.5f #define LAT_HOME_BASE 41.8902f /* Coast Guard Station Sector Alpha */ #define LON_HOME_BASE 12.4922f /* --- ENUMERATIONS --- */ typedef enum { VESSEL_UNKNOWN = 0x00, VESSEL_CIVILIAN = 0x01, VESSEL_COMMERCIAL = 0x02, VESSEL_MILITARY = 0x03, VESSEL_DISTRESS = 0xFF /* MAYDAY / Priority Target */ } VesselType; typedef enum { STATUS_OK, STATUS_WARN_OUT_OF_BOUNDS, STATUS_CRITICAL_MAYDAY } SystemStatus; /* --- DATA STRUCTURES --- */ typedef struct { unsigned int vessel_id; char callsign[8]; float latitude; float longitude; float speed_knots; float heading_deg; VesselType type; unsigned long timestamp_epoch; } RadarTarget; typedef struct { RadarTarget target_list[MAX_TARGETS]; unsigned int active_count; unsigned int alert_flag; } RadarSystemContext; /* ============================================================================ * MODULE 1: TELEMETRY & COORDINATE CALCULATIONS * ============================================================================ */ /** * Calculates straight-line distance (Nautical Miles approximation) * between station home base and target coordinates. */ float NAV_CalculateDistanceNM(float lat2, float lon2) { float dlat = (lat2 - LAT_HOME_BASE) * (3.14159265f / 180.0f); float dlon = (lon2 - LON_HOME_BASE) * (3.14159265f / 180.0f); /* Euclidean approximation optimized for CRT Workstations */ float a = sin(dlat/2.0f) * sin(dlat/2.0f) + cos(LAT_HOME_BASE * (3.14159265f / 180.0f)) * cos(lat2 * (3.14159265f / 180.0f)) * sin(dlon/2.0f) * sin(dlon/2.0f); return 2.0f * 3440.065f * atan2(sqrt(a), sqrt(1.0f - a)); } /* ============================================================================ * MODULE 2: RADAR TARGET MANAGEMENT & PROCESSING * ============================================================================ */ /** * Initializes System Context memory pool. */ void RADAR_InitSystem(RadarSystemContext *ctx) { if (ctx == NULL) return; ctx->active_count = 0; ctx->alert_flag = 0; memset(ctx->target_list, 0, sizeof(RadarTarget) * MAX_TARGETS); printf(">>> NAV-SENTINEL v3.81 INITIALIZED <<<\n"); printf("[SYS] SYSTEM RADAR OPERATIONAL - RANGE: %.1f NM\n", RADAR_RANGE_NM); printf("[SYS] BASE STATION COORD: %.4f, %.4f\n\n", LAT_HOME_BASE, LON_HOME_BASE); } /** * Registers new vessel telemetry into active tracking matrix. */ SystemStatus RADAR_RegisterTarget(RadarSystemContext *ctx, RadarTarget target) { if (ctx->active_count >= MAX_TARGETS) { printf("[WARN] MATRIX FULL. TARGET OVERFLOW.\n"); return STATUS_WARN_OUT_OF_BOUNDS; } float dist = NAV_CalculateDistanceNM(target.latitude, target.longitude); if (dist > RADAR_RANGE_NM) { printf("[TRACKING IGNORED] TARGET %s OUT OF SECTOR RANGE (%.2f NM)\n", target.callsign, dist); return STATUS_WARN_OUT_OF_BOUNDS; } ctx->target_list[ctx->active_count] = target; ctx->active_count++; if (target.type == VESSEL_DISTRESS) { ctx->alert_flag = 1; return STATUS_CRITICAL_MAYDAY; } return STATUS_OK; } /* ============================================================================ * MODULE 3: SEARCH AND RESCUE (SAR) & DISPATCH ROUTINES * ============================================================================ */ /** * Scans active target table for MAYDAY triggers and dispatches SAR alert. */ void SAR_EvaluateDistressSignals(const RadarSystemContext *ctx) { printf("--- [INSPECTING RADAR MATRIX: %u ACTIVE TARGETS] ---\n", ctx->active_count); for (unsigned int i = 0; i < ctx->active_count; i++) { const RadarTarget *v = &ctx->target_list[i]; float dist = NAV_CalculateDistanceNM(v->latitude, v->longitude); printf("ID: %04u | CALL: %-7s | SPD: %02.1f KTS | HDG: %03.0f° | DIST: %04.1f NM ", v->vessel_id, v->callsign, v->speed_knots, v->heading_deg, dist); if (v->type == VESSEL_DISTRESS) { printf("[*** MAYDAY ALERT ***]\n"); printf(">> EMERGENCY DISPATCH: COAST GUARD CUTTER INTERCEPT INITIATED <<\n"); printf(">> TARGET POSITION: LAT %.4f / LON %.4f\n", v->latitude, v->longitude); } else { printf("[NORMAL]\n"); } } printf("----------------------------------------------------\n\n"); } /* ============================================================================ * MAIN ENTRY POINT (SYSTEM SIMULATION) * ============================================================================ */ int main(void) { RadarSystemContext main_radar_sys; RADAR_InitSystem(&main_radar_sys); /* Target 1: Commercial Cargo */ RadarTarget t1 = { .vessel_id = 1001, .callsign = "MERCHANT", .latitude = 41.9500f, .longitude = 12.5500f, .speed_knots = 14.5f, .heading_deg = 180.0f, .type = VESSEL_COMMERCIAL, .timestamp_epoch = 910180800 /* Nov 1998 epoch */ }; /* Target 2: Vessel in Distress (MAYDAY) */ RadarTarget t2 = { .vessel_id = 9999, .callsign = "MAYDAY1", .latitude = 42.1000f, .longitude = 12.6000f, .speed_knots = 0.0f, .heading_deg = 0.0f, .type = VESSEL_DISTRESS, .timestamp_epoch = 910180850 }; /* Registering Telemetry */ RADAR_RegisterTarget(&main_radar_sys, t1); RADAR_RegisterTarget(&main_radar_sys, t2); /* Run SAR Engine Sweep */ SAR_EvaluateDistressSignals(&main_radar_sys); return 0; }
